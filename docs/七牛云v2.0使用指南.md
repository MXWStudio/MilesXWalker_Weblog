# 七牛云 v2.0 使用指南

## 🚀 快速开始

### 1. 配置环境变量

编辑项目根目录的 `.env` 文件：

```bash
# 七牛云密钥（服务端使用 - 不要提交到Git！）
QINIU_ACCESS_KEY=your_access_key_here
QINIU_SECRET_KEY=your_secret_key_here

# 存储空间配置
QINIU_BUCKET=your_bucket_name
QINIU_REGION=cn-east-1
QINIU_ZONE=z0

# 访问域名（前端需要）
VITE_QINIU_DOMAIN=https://cdn.example.com

# API 服务器地址
VITE_API_URL=http://localhost:3000

# 功能开关
VITE_ENABLE_QINIU=true
VITE_ENABLE_QINIU_DEBUG=true
```

### 2. 启动服务

```bash
# 方式1：同时启动前后端（推荐）
npm run start:dev

# 方式2：分别启动
npm run server:dev  # 后端服务器 (http://localhost:3000)
npm run dev         # 前端应用 (http://localhost:5173)
```

### 3. 验证配置

访问以下 URL 验证服务：

```bash
# 健康检查
http://localhost:3000/health

# 七牛云服务健康检查
http://localhost:3000/api/qiniu/health

# 七牛云连接测试
http://localhost:3000/api/qiniu/test

# API 文档
http://localhost:3000/api
```

## 📚 API 使用示例

### 基础文件上传

```javascript
import qiniuService from '@/services/qiniuService'

// 上传文件
async function uploadFile(file) {
  const result = await qiniuService.uploadFile(file, {
    // 可选参数
    fileName: 'custom-name.jpg', // 自定义文件名
    expires: 3600, // 凭证过期时间（秒）
    fsizeLimit: 10 * 1024 * 1024, // 文件大小限制（10MB）
    mimeLimit: 'image/*', // 文件类型限制

    // 进度回调
    onProgress: progress => {
      console.log(`上传进度: ${progress}%`)
    },
  })

  if (result.success) {
    console.log('上传成功！')
    console.log('文件 URL:', result.url)
    console.log('文件 Key:', result.key)
    console.log('文件 Hash:', result.hash)
    console.log('文件大小:', result.size)
  } else {
    console.error('上传失败:', result.error)
  }
}
```

### 大文件上传（分片）

```javascript
import { uploadLargeFile, smartUpload } from '@/utils/qiniuUploader'

// 方式1：手动使用分片上传
async function uploadLarge(file) {
  const result = await uploadLargeFile(file, {
    fileName: 'large-video.mp4',
    chunkSize: 4 * 1024 * 1024, // 分片大小（4MB）
    enableResume: true, // 启用断点续传
    maxRetries: 3, // 失败重试次数

    // 整体进度
    onProgress: ({ progress, uploadedChunks, totalChunks, uploadedBytes, totalBytes }) => {
      console.log(`总进度: ${progress}%`)
      console.log(`已上传: ${uploadedChunks}/${totalChunks} 个分片`)
      console.log(
        `已传输: ${(uploadedBytes / 1024 / 1024).toFixed(2)}MB / ${(totalBytes / 1024 / 1024).toFixed(2)}MB`
      )
    },

    // 单个分片完成
    onChunkComplete: ({ chunkIndex, totalChunks, uploadedChunks }) => {
      console.log(`分片 ${chunkIndex + 1}/${totalChunks} 上传完成`)
    },

    // 错误处理
    onError: ({ type, chunkIndex, error }) => {
      console.error(`错误类型: ${type}`)
      if (chunkIndex !== undefined) {
        console.error(`失败分片: ${chunkIndex}`)
      }
      console.error(`错误信息: ${error}`)
    },
  })

  return result
}

// 方式2：智能上传（自动判断是否使用分片）
async function smartUploadFile(file) {
  const result = await smartUpload(file, {
    threshold: 10 * 1024 * 1024, // 超过10MB使用分片上传
    enableResume: true,
    onProgress: ({ progress }) => {
      console.log(`上传进度: ${progress}%`)
    },
  })

  return result
}
```

### 批量上传

```javascript
// 批量上传多个文件
async function uploadMultipleFiles(files) {
  const result = await qiniuService.uploadFiles(files, {
    onProgress: progress => {
      console.log(`批量上传进度: ${progress}%`)
    },
  })

  console.log(`总计: ${files.length} 个文件`)
  console.log(`成功: ${result.successCount} 个`)
  console.log(`失败: ${result.failedCount} 个`)

  // 查看详细结果
  result.results.forEach((item, index) => {
    if (item.success) {
      console.log(`✅ ${files[index].name}: ${item.url}`)
    } else {
      console.error(`❌ ${files[index].name}: ${item.error}`)
    }
  })
}
```

### 文件管理操作

```javascript
// 获取文件信息
async function getFileInfo(key) {
  const result = await qiniuService.getFileInfo(key)

  if (result.success) {
    console.log('文件信息:', result.data)
    console.log('大小:', result.data.fsize)
    console.log('MIME类型:', result.data.mimeType)
    console.log('上传时间:', new Date(result.data.putTime / 10000))
  }
}

// 列举文件
async function listAllFiles(prefix = '') {
  const result = await qiniuService.listFiles({
    prefix: prefix, // 文件前缀（如 'uploads/'）
    limit: 100, // 每次返回数量
    marker: '', // 分页标记
  })

  if (result.success) {
    console.log('文件列表:')
    result.data.items.forEach(item => {
      console.log(`- ${item.key} (${(item.fsize / 1024).toFixed(2)}KB)`)
    })

    // 如果还有更多文件
    if (result.data.marker) {
      console.log('还有更多文件，使用 marker:', result.data.marker)
    }
  }
}

// 删除文件
async function deleteFile(key) {
  const result = await qiniuService.deleteFile(key)

  if (result.success) {
    console.log('文件已删除')
  } else {
    console.error('删除失败:', result.error)
  }
}

// 批量删除
async function batchDeleteFiles(keys) {
  const result = await qiniuService.batchDelete(keys)

  console.log(`成功删除: ${result.successCount} 个`)
  console.log(`删除失败: ${result.failedCount} 个`)

  // 查看详细结果
  result.results.forEach(item => {
    if (item.success) {
      console.log(`✅ ${item.key} 已删除`)
    } else {
      console.error(`❌ ${item.key} 删除失败: ${item.error}`)
    }
  })
}

// 移动/重命名文件
async function moveFile(oldKey, newKey) {
  const result = await qiniuService.moveFile(oldKey, newKey, false)

  if (result.success) {
    console.log(`文件已移动: ${oldKey} -> ${newKey}`)
  }
}

// 复制文件
async function copyFile(srcKey, destKey) {
  const result = await qiniuService.copyFile(srcKey, destKey, false)

  if (result.success) {
    console.log(`文件已复制: ${srcKey} -> ${destKey}`)
  }
}
```

### 简历云端存储

```javascript
import resumeCloudService from '@/services/resumeCloudService'

// 保存简历
async function saveResume(resumeData) {

    resumeId: 'my-resume-2024',
    onProgress: progress => {
      console.log(`保存进度: ${progress}%`)
    },
  })

  if (result.success) {
    console.log('简历已保存到云端')
    console.log('文件URL:', result.data.url)
    console.log('保存时间:', result.data.savedAt)
  }

// 加载简历
async function loadResume(resumeId = 'default') {
  const result = await resumeCloudService.loadResume(resumeId)

  if (result.success) {
    console.log('简历已加载')
    console.log('简历数据:', result.data)
    console.log('保存时间:', result.metadata.savedAt)
    console.log('版本:', result.metadata.version)

    return result.data
  } else {
    console.error('加载失败:', result.error)
  }

// 列举所有简历
async function listMyResumes() {
  const result = await resumeCloudService.listResumes()

  if (result.success) {
    console.log(`找到 ${result.data.length} 份简历:`)
    result.data.forEach(resume => {
      console.log(`- ID: ${resume.resumeId}`)
      console.log(`  大小: ${(resume.size / 1024).toFixed(2)}KB`)
      console.log(`  更新时间: ${resume.updatedAt}`)
      console.log(`  URL: ${resume.url}`)
    })
  }
}

// 删除简历
async function deleteResume(resumeId) {
  const result = await resumeCloudService.deleteResume(resumeId)

  if (result.success) {
    console.log('简历已从云端删除')
  }
}

// 自动保存（防抖）
function setupAutoSave(resumeData) {
  // 会在3秒无操作后自动保存

}

function cancelAutoSave() {
  resumeCloudService.cancelAutoSave()
}
```

## 🎨 Vue 组件示例

### 文件上传组件

```vue
<template>
  <div class="upload-component">
    <div class="upload-area" @dragover.prevent @drop.prevent="handleDrop">
      <input
        ref="fileInput"
        type="file"
        multiple
        @change="handleFileSelect"
        style="display: none"
      />

      <button @click="$refs.fileInput.click()">选择文件</button>

      <p>或拖拽文件到这里</p>
    </div>

    <!-- 上传进度 -->
    <div v-if="uploading" class="progress">
      <div class="progress-bar" :style="{ width: progress + '%' }"></div>
      <span>{{ progress }}%</span>
    </div>

    <!-- 上传结果 -->
    <div v-if="uploadedFiles.length" class="file-list">
      <h3>已上传文件：</h3>
      <div v-for="file in uploadedFiles" :key="file.key" class="file-item">
        <img v-if="file.isImage" :src="file.thumbnail" alt="" />
        <span>{{ file.name }}</span>
        <a :href="file.url" target="_blank">查看</a>
        <button @click="deleteFile(file.key)">删除</button>
      </div>
    </div>
  </div>
</template>

<script setup>
import { ref } from 'vue'
import { smartUpload } from '@/utils/qiniuUploader'
import qiniuService from '@/services/qiniuService'

const uploading = ref(false)
const progress = ref(0)
const uploadedFiles = ref([])

const handleFileSelect = async event => {
  const files = Array.from(event.target.files)
  await uploadFiles(files)
}

const handleDrop = async event => {
  const files = Array.from(event.dataTransfer.files)
  await uploadFiles(files)
}

const uploadFiles = async files => {
  for (const file of files) {
    uploading.value = true
    progress.value = 0

    const result = await smartUpload(file, {
      onProgress: ({ progress: p }) => {
        progress.value = p
      },
    })

    if (result.success) {
      const isImage = file.type.startsWith('image/')

      uploadedFiles.value.push({
        name: file.name,
        key: result.key,
        url: result.url,
        isImage,
        thumbnail: isImage
          ? qiniuService.getThumbnailUrl(result.key, {
              width: 100,
              height: 100,
            })
          : null,
      })
    }

    uploading.value = false
  }
}

const deleteFile = async key => {
  const result = await qiniuService.deleteFile(key)

  if (result.success) {
    uploadedFiles.value = uploadedFiles.value.filter(f => f.key !== key)
    alert('删除成功')
  } else {
    alert('删除失败: ' + result.error)
  }
}
</script>
```

## ⚠️ 注意事项

### 安全性

1. **不要将密钥提交到 Git**

   ```bash
   # .gitignore 应包含
   .env
   .env.local
   ```

2. **使用环境变量**

   ```bash
   # 生产环境使用不同的配置
   QINIU_ACCESS_KEY=prod_access_key
   QINIU_SECRET_KEY=prod_secret_key
   ```

3. **设置合理的凭证过期时间**
   ```javascript
   // 短期操作使用较短的过期时间
   await qiniuService.getUploadToken({ expires: 600 }) // 10分钟
   ```

### 性能优化

1. **大文件使用分片上传**

   ```javascript
   // 超过 10MB 的文件使用分片上传
   await smartUpload(file, { threshold: 10 * 1024 * 1024 })
   ```

2. **使用图片处理**

   ```javascript
   // 使用缩略图而不是原图
   const thumb = qiniuService.getThumbnailUrl(key, { width: 200 })
   ```

3. **启用断点续传**
   ```javascript
   // 大文件上传启用断点续传
   await uploadLargeFile(file, { enableResume: true })
   ```

### 错误处理

```javascript
try {
  const result = await qiniuService.uploadFile(file)

  if (result.success) {
    // 成功处理
  } else {
    // 失败处理
    console.error('上传失败:', result.error)

    // 向用户显示友好的错误消息
    alert(`上传失败: ${result.error}`)
  }
} catch (error) {
  // 异常处理
  console.error('发生异常:', error)
  alert('上传过程中发生错误，请重试')
}
```
