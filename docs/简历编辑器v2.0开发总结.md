### 未来 AI 功能实现思路

```javascript
/**
 * AI 定制简历（待实现）
 * @param {Object} resumeData - 原始简历数据
 * @param {String} targetJob - 目标岗位
 * @returns {Object} 优化后的简历数据
 */
async function customizeResumeWithAI(resumeData, targetJob) {
  // 1. 调用 AI API
  const response = await fetch('/api/ai/customize-resume', {
    method: 'POST',
    body: JSON.stringify({
      resumeData,
      targetJob,
    }),
  })

  // 2. 获取优化建议
  const optimized = await response.json()

  // 3. 返回优化后的简历
  return {
    ...resumeData,
    summary: optimized.summary, // 优化后的个人简介
    experience: optimized.experience, // 重新排序和优化
    projects: optimized.projects, // 突出相关项目
    skills: optimized.skills, // 推荐技能
  }
}
```

### 1. 技能列表的双格式支持

```javascript
// 在 ResumeEditor.vue 中
const skillsText = ref('') // 显示用文本

// 初始化
if (Array.isArray(resumeData.value.skills)) {
  skillsText.value = resumeData.value.skills.join('\n')
}

// 更新时转换
const updateSkills = event => {
  const text = event.target.value
  const skillsArray = text
    .split('\n')
    .map(s => s.trim())
    .filter(s => s.length > 0)

  resumeData.value.skills = skillsArray
}
```

```javascript
// 输入：逗号分隔字符串
```

```javascript
function generateFileName(resumeData) {
  const { fullName, title, targetJob } = resumeData
  const namePart = fullName || '简历'
  const jobPart = targetJob || title || '通用' // 优先使用目标岗位

  return `${namePart}-${jobPart}-${timestamp}`
```
